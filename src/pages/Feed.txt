const COLORS: string[] = ["#4F8EF7","#F7874F","#4FD1C5","#9F7AEA","#F687B3","#68D391","#FC8181"];

interface AvatarProps {
  name?: string;
  size?: number;
  src?: string;
  online?: boolean;
  ring?: boolean;
}

function Avatar({ name = "", size = 36, src, online = false, ring = false }: AvatarProps) {
  const initials = name.split(" ").map((w: string) => w[0]).join("").slice(0, 2).toUpperCase();
  const color = COLORS[name.charCodeAt(0) % COLORS.length];
  return (
    <div className="relative flex-shrink-0" style={{ width: size, height: size }}>
      <div
        className={`flex items-center justify-center rounded-full font-semibold text-white select-none ${ring ? "ring-2 ring-blue-500 ring-offset-1" : ""}`}
        style={{ width: size, height: size, background: src ? undefined : color, fontSize: size * 0.35 }}
      >
        {src ? <img src={src} alt={name} className="rounded-full object-cover w-full h-full" /> : initials}
      </div>
      {online && (
        <span className="absolute bottom-0 right-0 w-2.5 h-2.5 bg-green-400 rounded-full border-2 border-white" />
      )}
    </div>
  );
}

  return (

          <div key={i} className="w-5 h-5 rounded-full border-2 border-white flex items-center justify-center text-xs" style={{ background: c }}>
            <span style={{ fontSize: 10 }}>{i === 0 ? "😆" : i === 1 ? "❤️" : "👍"}</span>
          </div>
        ))}
      </div>
      {count > 3 && <span className="text-xs text-gray-500">{count}+</span>}
    </div>
  );
}
